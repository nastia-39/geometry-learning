\section{Introduction}


\begin{abstract}
	Transformers with billions of parameters have achieved remarkable success across various domains, such as language modeling and computer vision.
	%
	Existing works analyze the empirically observed phenomena, such as training dynamics, in-context learning (ICL) generalization, depth/width effects, scaling laws, only in isolation and in very restricted settings (single-layer, linear regression, fixed width), leaving open whether they are manifestations of a single underlying mechanism.
	%
	In this paper, we show how a Transformer-like Energy-Based Model (TEBM) can provably learn a Gibbs distribution over a function space $f \propto \exp(-\energy_{\theta}(f))$ from noisy observations $f_j(x_i) + \epsilon$.
	%
	We consider linear families of energy functionals and show how in the mean-field regime, where the number of heads $\to \infty$, training a TEBM recovers the parameter $\theta$ with provable approximation and generalization guarantees.
	%
	Our findings prove the convergence of the unsupervised training dynamics of TEBMs, the emergence of scaling laws in the considered setting, and the necessity of the regularization of $\theta$ for the generalization properties of the ICL.
	%
	We evaluate our architecture on synthetic and real-world benchmarks to support our theoretical findings.
\end{abstract}

% 	\begin{align*}
% 		error = n^{-\alpha} + m^{-\beta} + H^{-\gamma} + D^{-\delta} + T^{-\tau}
% 	\end{align*}
% \end{theorem}

This paper makes the following contributions:
\begin{itemize}
	\item \textbf{Learning distributions over functions.}
	      We formulate the problem of learning a distribution $\nu$ over a function space
	      $\ffam$ from noisy observations as a regularized optimization problem.
	      We model $\nu$ as an exponential family with sufficient statistic $\Phi \colon \ffam \to \Real^d$,
	      so that $p(f) \propto \exp(-\langle \theta, \Phi(f) \rangle)$,
	      and state generalization guarantees for several (semi-)parametric choices of $\Phi$.

	\item \textbf{Connection to in-context learning.}
	      We draw a connection of this problem to the in-context learning in Transformer-like Energy-Based Models, discuss how it can explain the convergence to the optimal solution, the empirically observed scaling laws, the role of the positional encodings and the importance of the regularization for the generalization properties.

	\item \textbf{Architectural implications and empirical validation.}
	      Guided by the optimization problem, we propose a modification of the Transformer
	      architecture that enforces the structural assumptions of our framework.
	      Experiments on both synthetic and real-world benchmarks demonstrate improvements
	      in size-generalization, interpretability, and computational efficiency.
\end{itemize}

\section{Problem Setup}\label{sec:set-up}

We are given a manifold $\domain$ equipped with a probability measure $\mu$,
a function space $\ffam$ of functions $f \colon \domain \to \Real$,
endowed with an \emph{unknown} probability measure $\nu$ over $\ffam$.
We observe a dataset $\dataset = \left\{ D^{(1)}, \ldots, D^{(\nfunctions)} \right\}$ consisting of $\nfunctions$ observations $D^{(j)}$ of functions $f_1, \ldots, f_{\nfunctions} \overset{\text{i.i.d.}}{\sim} \nu(\ffam)$,
each evaluated at $\nsamples$ points drawn from $\mu$:
\begin{gather*}
	x_1, \ldots, x_{\nsamples} \overset{\text{i.i.d.}}{\sim} \mu(\domain), \qquad
	\epsilon_{j,i} \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, \sigma^2), \\
	\dataset = \Bigl\{
	\bigl\{ (x_i,\, f_j(x_i) + \epsilon_{j,i}) \bigr\}_{i=1}^{\nsamples}
	\Bigr\}_{j=1}^{\nfunctions}.
\end{gather*}

Our goal is to learn the measure $\nu$ over $\ffam$.
Given $\nu$, predicting a new function $f \sim \nu$ from noisy observations
$\{(x_i, y_i)\}_{i=1}^{\nsamples}$, where $y_i = f(x_i) + \epsilon_i$ and
$\epsilon_i \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, \sigma^2)$,
reduces to MAP estimation:
\begin{align*}
	\hat{f}_m
	= \arg\min_{f \in \ffam}
	\sum_{i=1}^{\nsamples} \bigl| y_i - f(x_i) \bigr|^2
	- \log \nu(f).
\end{align*}


\paragraph{Connection to ICL in Transformers.}

The setup described above is closely related to in-context learning in Transformers.
Given a prompt $(x_1, y_1), \ldots, (x_{\nsamples}, y_{\nsamples})$, where $x_i$ are positional encodings
and $y_i$ are tokens, the task is to predict the next token at position $x_{\nsamples+1}$.
Each prompt can be viewed as a noisy observation of a latent function $f \sim \nu(\ffam)$,
and distinct prompts correspond to distinct draws from $\nu(\ffam)$.
The Transformer thus implicitly learns the shared structure across functions,
that is, the measure $\nu$ itself, and uses it to generalize within a new context.

\begin{assumption}[Gibbs distribution]
	We assume that the probability measure $\nu$ over $\ffam$ is a Gibbs distribution induced by a convex energy functional $E_{\theta}: \ffam \to \Real^+$ and a reference measure (\todo{define})$\nu_0$ on $\ffam$:
	\[
		\nu(f) = \frac{1}{Z_{\theta}} \exp(-E_{\theta}(f))

\end{assumption}

In this section, we discuss how to approximate our intractable convex optimization problem \eqref{eq:loss-w-data} by a bi-level non-convex optimization problem via masking approach. Fix a parameter $\theta$ and a dataset $D=\{(x_j,y_j)\}_{j=1}^n$.
Define
$
	L_\theta(f, D)
	:=
	\ell(D,f)+\lambda \energy_{\theta}(f),
$
where $\ell(D,f)$ denotes the empirical negative log-likelihood and
$\energy_{\theta}$ is an energy functional.

% Assume the following:
% \begin{enumerate}
% 	\item The dataset $D$ is large (large $n$) and noise variance $\sigma^2$ is small
% 	\item (\emph{Coercivity}) The functional $R_\theta$ is coercive on $X$
% 	      modulo a finite-dimensional nullspace $\mathcal N$, i.e.,
% 	      $\|f\|_X\to\infty$ with $f\perp\mathcal N$ implies $R_\theta(f)\to\infty$.
% 	\item (\emph{Strict convexity}) The functional $\ell(D,\cdot)$ is strictly

% \end{enumerate}

% Under these assumptions, 
Note, for a fixed $\theta$ the minimizing $L_\theta(f, D)$ with respect to $f$ is unique:
$
	\hat f_\theta
	=
	\arg\min_{f\in \ffam} L_\theta(f, D)
$
% admits a unique solution modulo $\mathcal N$.
Moreover, the posterior distribution
$
	p_\theta(f\mid D)
	\propto
	\exp\!\big(-L_\theta(f, D)\big)
$
concentrates in probability around $\hat f_\theta$ as either
$n\to\infty$ or $\sigma^2 \to 0$.
In particular, for any neighborhood $U$ of $\hat f_\theta$ in $\ffam$, $p_\theta(U\mid D) \longrightarrow 1$.

\paragraph{The Inner Problem.}
For a dataset $D = \{(x_i, y_i)\}$ we have
\begin{align*}
	\hat f_\theta = \argmin_f
	\log p\left(D \mid f\right)
	+
	\energy_{\theta}\left(f\right)
	=
	\argmin_f
	\| f(x_i) - y_i\|_2^2
	+
	\langle \theta, \Phi(f) \rangle
\end{align*}

Note, that due to convexity, this can be solved by subgradient descent on $f$ with a step size $\eta > 0$:
\begin{align*}
	f^{(t+1)} = f^{(t)} - \eta \left[ 2\sum_{i=1}^{\nsamples} \bigl(f^{(t)}(x_i) - y_i\bigr) \delta_{x_i} + \nabla_f \energy_{\theta}(f^{(t)}) \right]
\end{align*}
where $\nabla_f \energy_{\theta}(f^{(t)})$ is a (sub)gradient of the energy functional.

\section{Connection to Transformers And Its Implications}

In most of the examples above, the key step of implementing a (sub)gradient descent for the inner problem is to compute the discrete (anisotropic) Laplacian $\Delta_Q$ for some PSD matrix $Q$. The original Laplacian $\Delta_Q$ can be approximated by its discrete version $\hat \Delta_Q \in \Real^{n \times n}$ computed on a set of points $x_1, \ldots, x_n \in \domain$:
\begin{gather*}
	% \label{eq:discrete-laplacian}
	\hat \Delta_Q = I - D^{-1/2} W D^{-1/2} \quad \text{where} \\
	W_{ij} = \exp\left(\frac{-\langle x_i - x_j, \Pi_{x_i} Q^\dagger \Pi_{x_i} (x_i - x_j)\rangle}{\epsilon} \right) \quad \text{and} \quad D_{ii} = \sum_{j=1}^n W_{ij}.
\end{gather*}
